192        """Listen for proposals or rejections from trucks."""
193        async def run(self):
194            """Receive messages and store proposals in the agent's memory."""
195            msg = await self.receive(timeout=1)
196            if msg:
197                if msg.metadata.get("performative") == "propose": # Extracts path, cost and available capacity from the truck's proposal
198                    path_str, estimated_cost, available_capacity, fuel, truck_waste_type = msg.body.split(";")
199                    estimated_cost = int(estimated_cost)
200                    available_capacity = int(available_capacity)
201                    # Stores the proposal, including the path
202                    self.agent.received_responses[msg.sender] = {
203                        'type': 'proposal',
204                        'cost': estimated_cost,
205                        'available_capacity': available_capacity,
206                        'fuel': fuel,
207                        'path': path_str,
208                        'truck_waste_type': truck_waste_type
209                    }
210                    print(f"[{self.agent.name}] [{self.agent.waste_type}] Proposta recebida de {msg.sender}: Capacidade {available_capacity}, Custo {estimated_cost}, Tipo {truck_waste_type}")
211                elif msg.metadata.get("performative") == "decline":
212                    # Stores the rejection as a response
213                    self.agent.received_responses[msg.sender] = {'type': 'decline'}
214                    print(f"[{self.agent.name}] [{self.agent.waste_type}] Rejeição recebida de {msg.sender}")

217    class WaitForResponsesBehaviour(OneShotBehaviour):
218        """Wait for a fixed time to collect proposals and evaluate the best one."""
219        async def run(self):
220            """Wait 3 seconds and then trigger proposal evaluation."""
221            await asyncio.sleep(3) # wait for trucks responses to be sent
222            # Ends the wait when time expires and evaluates proposals
223            print(f"[{self.agent.name}] [{self.agent.waste_type}] Tempo de espera expirado. Avaliando propostas...")
224            await self.evaluate_proposals() # calls the function to choose the best proposal
225            self.agent.received_respondes = {}
226            self.kill()  # Ends the behavior after evaluation
227
228        async def evaluate_proposals(self):
229            """Select the best proposal based on cost, capacity and fuel level."""
230            best_proposal = None
231            for truck_jid, response in self.agent.received_responses.items():
232                if response['type'] == 'proposal':
233                    # get the right truck by its jid
234                    for t in self.agent.environment.trucks:
235                        if(t.jid == truck_jid):
236                            cost, path = t.get_shortest_path(self.agent.position)
237                            is_busy = t.is_busy                   
238                            # updates the proposal if meanwhile the truck moved during exploration
239                            if(is_busy==False and (cost!=response['cost'] or path!=ast.literal_eval(response['path']))):
240                                response['cost']=cost
241                                response['path']=f"{path}"
242                    
243                    # Check if truck waste type is compatible with bin waste type
244                    is_compatible = self.is_waste_type_compatible(response['truck_waste_type'], self.agent.waste_type)
245                    
246                    if (not is_busy and is_compatible and
247                        (best_proposal is None or                        
248                        response['available_capacity'] > best_proposal['available_capacity'] or
249                        (response['available_capacity'] == best_proposal['available_capacity'] and response['cost'] < best_proposal['cost']) or
250                        (response['available_capacity'] == best_proposal['available_capacity'] and response['cost'] == best_proposal['cost'] and response['fuel'] > best_proposal['fuel']) or
251                        (response['available_capacity'] == best_proposal['available_capacity'] and response['cost'] == best_proposal['cost'] and response['fuel'] > best_proposal['fuel']) and response['truck_jid'] < best_proposal['fuel'])):
252                        # Updates the best proposal based on capacity and cost
253                        best_proposal = {
254                            'truck_jid': truck_jid,
255                            'cost': response['cost'],
256                            'available_capacity': response['available_capacity'],
257                            'fuel': response['fuel'],
258                            'path': response['path'],
259                            'truck_waste_type': response['truck_waste_type']
260                        }
261
262            if best_proposal:
263                # If there is a valid proposal, send acceptance
264                if self.agent.current_waste >= 70:
265                    # MARKS THAT THIS COLLECTION WILL BE BY CFP
266                    self.agent.last_collection_type = 'cfp'
267                    await self.accept_best_proposal(best_proposal)
268                else:
269                    self.agent.is_waiting_for_truck = False
270                    self.agent.sent_colection_request = False

274        """Listen for problem reports from trucks."""
275        # it is cyclic because it is always listening for messages
276        async def run(self):
277            """Check for 'problem' messages and initiate wait logic."""
278            msg = await self.receive(timeout = 1)
279            if msg and msg.metadata.get("performative") == "problem": # truck warns bin that there was a problem (e.g., truck that was going to collect that waste got stuck in traffic or broke down)
280                # Adds a waiting behavior to collect responses from trucks
281                wait_for_problem_responses = self.agent.WaitForProblemResolveBehaviour() # bin creates a waiting behavior
282                self.agent.add_behaviour(wait_for_problem_responses) #adds the behavior to itself to start the time count
283                print(f"[{self.agent.name}] [{self.agent.waste_type}] O bin detetou problema")

338            
339            # Reset flag only when waste stops decreasing
340            if self.agent.current_waste >= self.agent.last_waste_level:

260                            'truck_waste_type': response['truck_waste_type']
261                        }

263            if best_proposal:

271                    self.agent.sent_colection_request = False

315        async def run(self):
316            """Monitor waste levels to detect collections and record times."""
317            # Detect any decrease in waste (not just >5)

320            if waste_decreased and not self.agent.collection_counted:

324                if self.agent.last_collection_type == 'cfp':

330                        collection_duration = self.agent.environment.timer() - self.agent.request_start_time
331                        self.agent.collection_time.append(collection_duration)
332                        print(f"[{self.agent.name}] CFP collection completed in {collection_duration}s. Times: {self.agent.collection_time}")
333

339            
340            # Reset flag only when waste stops decreasing
341            if self.agent.current_waste >= self.agent.last_waste_level:
342                self.agent.collection_counted = False

344                if self.agent.current_waste < 0.7 * self.agent.max_capacity:

346
