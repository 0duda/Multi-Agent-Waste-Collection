 79    def expand_aux(self, grid, row, col, dir_x, dir_y):
 80        """
 81        Helper method to connect adjacent nodes in the graph.
 82
 83        Args:
 84            grid (list): The simulation grid.       
 85            row (int): Current row.         
 86            col (int): Current column.      
 87            dir_x (int): Direction offset for X.        
 88            dir_y (int): Direction offset for Y.        
 89        """
 90        n_col = col + dir_x
 91        n_row = row + dir_y
 92        if 0 <= n_col < len(grid[0]) and 0 <= n_row < len(grid):
 93            if grid[n_row][n_col] <= 2: # creates edges between adjacent nodes (up, down, left, right)
 94                self.g.add_node(self.node_name_template(n_row, n_col))
 95                if not self.g.has_edge(self.node_name_template(row, col), self.node_name_template(n_row, n_col)):
 96                    self.g.add_edge(self.node_name_template(row, col), self.node_name_template(n_row, n_col), weight=1) # standard travel cost = 1
 97        return

113    def convert_to_graph(self):
114        """Convert the 2D grid into a NetworkX Graph with nodes and edges for navigable areas."""
115        grid = copy.deepcopy(self.grid)
116        for row in range(len(grid)):
117            for col in range(len(grid[0])):
118                if grid[row][col] <= 2:    #any number below or equal to one represents the road or bins(obstacles are ignored)
119                    grid[row][col] = -1
120                    self.g.add_node(self.node_name_template(row, col))
121                    self.expand(grid, row, col, [[1, 0], [-1, 0], [0, 1], [0, -1]])

258        """
259        Find bins close to a given position, ordered by distance.
260
261        Args:
262            position (tuple): The reference position (x, y).        
263            waste_type (str, optional): Filter by waste type.
264
265        Returns:
266            list: A list of nearby BinAgent objects.
267        """
268        nearby_bins = []
269        for bin in self.bins.values():
270            # Filter by waste type if specified
271            if waste_type and bin.waste_type != waste_type:
272                continue
273                
274            if bin.current_waste >= bin.max_capacity*0.4:
275                # Calculate distance (or other proximity criteria)
276                distance = abs(bin.position[0] - position[0]) + abs(bin.position[1] - position[1])
277                if distance <= 15:  # Exploration radius defined as 15
278                    nearby_bins.append((bin, distance))
279        sorted_bins = sorted(nearby_bins, key=lambda x: x[1])
280        # Returns only the bins, discarding distances
281        return [bin for bin, _ in sorted_bins]

319        """

337    async def set_traffic(self, level):
338        """
339        Apply traffic congestion to the graph edges.
340        
341        Args:
342            level (int): Traffic severity (0 = none, 1-5 = increasing congestion).
343        """
344        match level:
345            case 0:
346                edges=list(self.g.edges())
347                for edge_name in edges:
348                    self.g.edges[edge_name]['weight'] = 1
349                return
350            case 1:
351                percentage=0.3 # percentage of edges affected by traffic
352                multiplier=1.5 # how much the edge weight increases, simulating slowness
353            case 2:
354                percentage=0.3
355                multiplier=2
356            case 3:
357                percentage=0.5
358                multiplier=2
359            case 4:
360                percentage=0.6
361                multiplier=2.5
362            case 5:
363                percentage=0.8
364                multiplier=3
365            case _:
366                print("Invalid level of traffic, no traffic applied")
367                return
368            
369        edges=list(self.g.edges)
370        n_edges=int(len(edges)*percentage)
371
372        for _ in range(n_edges):
373            idx=randint(0,len(edges)-1) # chooses edges randomly
374            edge_name=edges[idx]
375            current_weight= self.g.edges[edge_name].get('weight')
376            self.g.edges[edge_name]['weight']= int(current_weight*multiplier)
377        
378        self.sendEnvironmentUpdate()

436        Calculate system statistics by waste type.

440        """

452        # Calculate average waste levels
453        for waste_type in stats:
454            if stats[waste_type]['bins'] > 0:
455                stats[waste_type]['avg_waste_level'] = stats[waste_type]['total_waste'] / stats[waste_type]['bins']
456        
457        # Count trucks by type
458        for truck in self.trucks:
459            if truck.truck_type in stats:
460                stats[truck.truck_type]['trucks'] += 1

477async def load_environment_from_json(file_path):
478    """
479    Load environment configuration and agents from a JSON file.
480
481    Args:
482        file_path (str): Path to the .json configuration file.
483
484    Returns:
485        Environment: A fully configured Environment instance.
486    """
487    with open(file_path, 'r') as f:
488        data = json.load(f)
489
490    env_config = data["environment"]
491    rows = env_config["rows"]
492    cols = env_config["cols"]
493    layout = env_config["layout"]
494
495    env = Environment(cols, rows, layout)
496
497    # Traffic from JSON 
498    traffic_level = env_config.get("traffic", None)
499    if traffic_level is not None:
500        print(f"🚦 Setting traffic level {traffic_level} from JSON...")
501        await env.set_traffic(traffic_level)
502
503    # Central
504    if "central" in data:
505        central_pos = data["central"]
506        env.central = (central_pos[1], central_pos[0])
507        env.grid[env.central[0]][env.central[1]] = 2
508        print(f"Central set to {env.central}")
509
510    # Roadblocks
511    for rb in data.get("roadblocks", []):
512        rb_position = (rb[1], rb[0])
513        await env.add_roadBlock(rb_position)
514
515    # Bins 
516    for b in data.get("bins", []):
517        position = (b["position"][1], b["position"][0])
518        waste_type = b.get("type", "organic")
519        max_capacity = b.get("max_capacity", 100)  # Default value 100
520
521        await env.add_bin(position, waste_type=waste_type)
522        bin_agent = env.get_bin_at_position(position)
523        if bin_agent:
524            bin_agent.max_capacity = max_capacity
525            print(f"✅ Bin at ({position[1]},{position[0]}) set to max_capacity: {max_capacity}")
526
527    # Trucks
528    for t in data.get("trucks", []):
529        position = (t["position"][1], t["position"][0])
530        truck_type = t.get("type", "organic")
531        max_load = t.get("max_load", 400)  # Default value 400
532        max_fuel = t.get("max_fuel", 100)  # Default value 100
533
534        await env.add_truck(position, truck_type=truck_type)
535        truck = env.trucks[-1]
536        truck.max_load = max_load
537        truck.max_fuel = max_fuel
538        truck.fuel = max_fuel
539        print(f"Truck at ({position[1]},{position[0]}) set to max_load: {max_load}, max_fuel: {max_fuel}")
540
541    print("Environment loaded successfully!")
542    return env

258    def get_nearby_bins(self, position, waste_type=None):
259        """
260        Find bins close to a given position, ordered by distance.
261
262        Args:
263            position (tuple): The reference position (x, y).        
264            waste_type (str, optional): Filter by waste type.
265
266        Returns:
267            list: A list of nearby BinAgent objects.
268        """
269        nearby_bins = []
270        for bin in self.bins.values():

272            if waste_type and bin.waste_type != waste_type:
273                continue
274                
275            if bin.current_waste >= bin.max_capacity*0.4:
276                # Calculate distance (or other proximity criteria)

279                    nearby_bins.append((bin, distance))

339        """
340        Apply traffic congestion to the graph edges.
341        
342        Args:
343            level (int): Traffic severity (0 = none, 1-5 = increasing congestion).
344        """
345        match level:
346            case 0:
347                edges=list(self.g.edges())
348                for edge_name in edges:
349                    self.g.edges[edge_name]['weight'] = 1
350                return
351            case 1:
352                percentage=0.3 # percentage of edges affected by traffic
353                multiplier=1.5 # how much the edge weight increases, simulating slowness
354            case 2:
355                percentage=0.3
356                multiplier=2
357            case 3:
358                percentage=0.5
359                multiplier=2
360            case 4:
361                percentage=0.6
362                multiplier=2.5
363            case 5:
364                percentage=0.8
365                multiplier=3
366            case _:
367                print("Invalid level of traffic, no traffic applied")
368                return
369            
370        edges=list(self.g.edges)
371        n_edges=int(len(edges)*percentage)
372
373        for _ in range(n_edges):
374            idx=randint(0,len(edges)-1) # chooses edges randomly
375            edge_name=edges[idx]
376            current_weight= self.g.edges[edge_name].get('weight')
377            self.g.edges[edge_name]['weight']= int(current_weight*multiplier)
378        
379        self.sendEnvironmentUpdate()
